export const selectCode = `import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@chenyibo111/ui"

<Select value={priority} onValueChange={setPriority}>
  <SelectTrigger aria-label="处理优先级">
    <SelectValue placeholder="选择优先级" />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="high">高</SelectItem>
    <SelectItem value="medium">中</SelectItem>
    <SelectItem value="low">低</SelectItem>
  </SelectContent>
</Select>`

`Select` 由 Trigger、Content 和 Item 组合，适合由组件库统一拥有弹层几何与键盘交互的固定选项。

## 何时不用

不要把 `Select` 当作原生 `<select>` 的样式替代品来猜测菜单打开逻辑。需要自由输入、远程搜索或大量选项时，应使用专门的 Combobox/搜索方案。

## 可访问性

- 触发器可通过 ArrowDown、ArrowUp、Enter 与 Escape 完成键盘操作。
- 不要只用 placeholder 命名选择器；使用 `aria-label` 或 `Field` 标签。
