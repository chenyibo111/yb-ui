import { ToastTriggerExample } from "../../../../components/examples/toast/trigger"

export const toastCode = `import { useState } from "react"
import { Button, Toast, ToastProvider, ToastTitle, ToastViewport } from "@chenyibo111/ui"

const [open, setOpen] = useState(false)

<ToastProvider>
  <Button onClick={() => setOpen(true)}>显示保存提示</Button>
  <Toast open={open} onOpenChange={setOpen}>
    <ToastTitle>项目已保存</ToastTitle>
  </Toast>
  <ToastViewport />
</ToastProvider>`

Toast 由应用明确控制打开状态，并通过统一的 Viewport 展示。它适合“已保存”“已复制”等可短暂确认的结果。

## 何时不用

Toast 不应承载唯一的错误原因或恢复指引。用户必须修正的问题应保留在 Field 或 Alert 中。

## 可访问性

- 保持标题短而明确，避免连续重复触发相同 Toast。
- 关键错误和必需操作不要只放在会自动消失的 Toast 中。
